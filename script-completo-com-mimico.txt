-- --- LÓGICA DE SELEÇÃO DE JOGADORES (SPECTATE) ---
local function GetOtherPlayers()
    local playerList = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player then
            table.insert(playerList, p)
        end
    end
    return playerList
end

local function GetSelectedPlayer()
    local otherPlayers = GetOtherPlayers()
    if #otherPlayers == 0 then return nil end
    if selectedPlayerIndex > #otherPlayers then selectedPlayerIndex = 1 end
    if selectedPlayerIndex < 1 then selectedPlayerIndex = #otherPlayers end
    return otherPlayers[selectedPlayerIndex]
end

local function UpdatePlayerLabel()
    local target = GetSelectedPlayer()
    if target then
        PlayerSelectLabel.Text = "Alvo: " .. target.DisplayName
    else
        PlayerSelectLabel.Text = "Alvo: Nenhum Player"
    end
end

BtnNextPlayer.MouseButton1Click:Connect(function()
    local otherPlayers = GetOtherPlayers()
    if #otherPlayers > 0 then
        selectedPlayerIndex = (selectedPlayerIndex % #otherPlayers) + 1
        UpdatePlayerLabel()
        if isSpectating then
            local target = GetSelectedPlayer()
            if target and target.Character and target.Character:FindFirstChild("Humanoid") then
                Camera.CameraSubject = target.Character.Humanoid
            end
        end
    end
end)

BtnPrevPlayer.MouseButton1Click:Connect(function()
    local otherPlayers = GetOtherPlayers()
    if #otherPlayers > 0 then
        selectedPlayerIndex = selectedPlayerIndex - 1
        if selectedPlayerIndex < 1 then selectedPlayerIndex = #otherPlayers end
        UpdatePlayerLabel()
        if isSpectating then
            local target = GetSelectedPlayer()
            if target and target.Character and target.Character:FindFirstChild("Humanoid") then
                Camera.CameraSubject = target.Character.Humanoid
            end
        end
    end
end)

-- Loop principal do Mímico (Posição + Animações)
RunService.Heartbeat:Connect(function()
    if not isMimicEnabled then return end
    if not selectedMimicTarget or not selectedMimicTarget.Character then return end
    
    local targetChar = selectedMimicTarget.Character
    local myChar = Player.Character
    if not targetChar or not myChar then return end

    local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    local targetHumanoid = targetChar:FindFirstChildOfClass("Humanoid")
    local myHumanoid = myChar:FindFirstChildOfClass("Humanoid")
    
    if not targetRoot or not myRoot or not targetHumanoid or not myHumanoid then return end

    local distance = 3
    local offsetCF = CFrame.new(0, 0, distance)

    if mimicOffsetType == "Frente" then
        offsetCF = CFrame.new(0, 0, -distance)
    elseif mimicOffsetType == "Esquerda" then
        offsetCF = CFrame.new(-distance, 0, 0)
    elseif mimicOffsetType == "Direita" then
        offsetCF = CFrame.new(distance, 0, 0)
    end

    myRoot.CFrame = targetRoot.CFrame * offsetCF
    myRoot.Velocity = targetRoot.Velocity

    if targetHumanoid.Jump and not myHumanoid.Jump then
        myHumanoid.Jump = true
    end

    local animator = targetHumanoid:FindFirstChildOfClass("Animator")
    local myAnimator = myHumanoid:FindFirstChildOfClass("Animator")
    
    if animator and myAnimator then
        local playingTracks = animator:GetPlayingAnimationTracks()
        if #playingTracks > 0 then
            local targetTrack = playingTracks[1]
            if targetTrack then
                local animId = targetTrack.Animation.AnimationId
                
                if not activeAnimTrack or activeAnimTrack.Animation.AnimationId ~= animId then
                    if activeAnimTrack then
                        activeAnimTrack:Stop()
                    end
                    local newAnim = Instance.new("Animation")
                    newAnim.AnimationId = animId
                    activeAnimTrack = myAnimator:LoadAnimation(newAnim)
                    activeAnimTrack:Play()
                    activeAnimTrack.TimePosition = targetTrack.TimePosition
                else
                    if math.abs(activeAnimTrack.TimePosition - targetTrack.TimePosition) > 0.1 then
                        activeAnimTrack.TimePosition = targetTrack.TimePosition
                    end
                end
            end
        else
            if activeAnimTrack then
                activeAnimTrack:Stop()
                activeAnimTrack = nil
            end
        end
    end
end)
